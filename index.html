Connect Google Sheets
1. Open a new Google Sheet
2. Go to Extensions → Apps Script
3. Paste the script below and click Deploy → New deployment → Web app → Anyone → Deploy
4. Copy the Web App URL and paste it here
function doPost(e){
  var sheet=SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d=JSON.parse(e.postData.contents);
  if(sheet.getLastRow()===0)sheet.appendRow(['Candidate','Phone','Role','Company','Methods','Mode','Date']);
  sheet.appendRow([d.name,d.phone,d.role,d.co,d.methods,d.mode,d.date]);
  return ContentService.createTextOutput('ok');
}
